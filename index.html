// 1. 初始化地圖（中心點：新北市政府附近） const map = L.map('map').setView([25.0115, 121.4650], 15); // 2. 建立底圖 (WMTS) const basicLayer = L.tileLayer('https://wmts.nlsc.gov.tw/wmts/EMAP/default/EPSG:3857/{z}/{y}/{x}', { maxZoom: 19, attribution: '© 國土測繪圖資服務雲' }).addTo(map); const photoLayer = L.tileLayer('https://wmts.nlsc.gov.tw/wmts/PHOTO2/default/EPSG:3857/{z}/{y}/{x}', { maxZoom: 19, attribution: '© 國土測繪圖資服務雲' }); // 3. 建立「都市計畫土地使用分區」圖層 (使用 WMS 服務) // 註：採用內政部國土管理署/城鄉發展分區公開 WMS 服務 const urbanPlanLayer = L.tileLayer.wms('https://zonemap.tcd.gov.tw/kml/wms.aspx', { layers: 'UrbanPlan', // 都市計畫圖層代碼 format: 'image/png', transparent: true, // 設為透明以便疊加在底圖上 opacity: 0.65, // 設定透明度 65% attribution: '© 內政部國土管理署城鄉發展分區' }); // 4. 建立地籍圖層 const cadastralLayer = L.tileLayer('https://wmts.nlsc.gov.tw/wmts/CADASTRAL/default/EPSG:3857/{z}/{y}/{x}', { maxZoom: 19, opacity: 0.7, attribution: '© 國土測繪圖資服務雲' }); // 5. 設定控制選單 const baseMaps = { "臺灣通用電子地圖": basicLayer, "正射影像圖": photoLayer }; const overlayMaps = { "都市計畫使用分區": urbanPlanLayer, // 勾選後即可顯示都市計畫顏色分區 "地籍圖": cadastralLayer }; // 將控制項加入右上角 L.control.layers(baseMaps, overlayMaps, { collapsed: false }).addTo(map); // 預設直接開啟都市計畫圖層測試 urbanPlanLayer.addTo(map);
